\section{rootfs/etc/init.d/rcS}

\begin{lstlisting}[language=bash]
#!/bin/sh
# Veilbox init script
mount -t proc none /proc
mount -t sysfs none /sys
mount -t devtmpfs none /dev
mkdir -p /dev/pts
mount -t devpts none /dev/pts
mount -t tmpfs none /tmp

hostname -F /etc/hostname

# Mount state partition
STATE_DEV=""
if [ -b /dev/vdb ]; then
    STATE_DEV=/dev/vdb
elif [ -b /dev/sdb ]; then
    STATE_DEV=/dev/sdb
fi
if [ -n "$STATE_DEV" ]; then
    mount "$STATE_DEV" /mnt/state
fi

ifconfig lo 127.0.0.1 up
ifconfig eth0 0.0.0.0 up
udhcpc -i eth0 -b

syslogd -O /var/log/messages -s 64
containerd -c /etc/containerd/config.toml &
dropbear -F -R -p 22 &
\end{lstlisting}
\clearpage

\section{GRUB Rootless Installer (Python)}

\begin{lstlisting}[language=python]
#!/usr/bin/env python3
# Rootless GRUB installer for Veilbox
import struct
import sys

def install_grub(disk_path, boot_img_path, core_img_path):
    with open(boot_img_path, 'rb') as f:
        boot_img = f.read()
    with open(core_img_path, 'rb') as f:
        core_img = f.read()

    with open(disk_path, 'r+b') as f:
        # Save partition table
        f.seek(0x1BE)
        pt = f.read(64)

        # Write boot.img to LBA 0
        f.seek(0)
        f.write(boot_img)

        # Restore partition table
        f.seek(0x1BE)
        f.write(pt)

        # Write core.img at LBA 1
        f.seek(512)
        f.write(core_img)

    print(f"GRUB installed: boot.img ({len(boot_img)}B), "
          f"core.img ({len(core_img)}B)")
\end{lstlisting}
